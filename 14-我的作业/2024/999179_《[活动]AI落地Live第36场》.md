import asyncio

import aiohttp

import os

import json

import time

import hashlib

import re

from datetime import datetime

from playwright.async_api import async_playwright

import base64

from urllib.parse import urlparse, parse_qs

class WeixinVideoScraper:

def __init__(self):

self.download_dir = "downloaded_videos"

self.cookies_file = "cookies.json"

os.makedirs(self.download_dir, exist_ok=True)

self.base_url = "https://channels.weixin.qq.com/"

self.browser = None

self.context = None

self.page = None

async def init_browser(self):

"""初始化浏览器"""

playwright = await async_playwright().start()

self.browser = await playwright.chromium.launch(

headless=False, # 显示浏览器界面

args=['--start-maximized']

)

self.context = await self.browser.new_context(

viewport={'width': 1920, 'height': 1080}

)

self.page = await self.context.new_page()

# 添加请求拦截

await self.page.route("**/*", self.handle_route)

return playwright

async def handle_route(self, route, request):

"""处理请求拦截"""

if "video" in request.url.lower():

print(f"检测到视频请求: {request.url}")

self.save_video_url(request.url)

await route.continue_()

def save_video_url(self, url):

"""保存视频URL到文件"""

with open("video_urls.txt", "a", encoding="utf-8") as f:

f.write(f"{url}\n")

async def login(self):

"""登录微信视频号"""

await self.page.goto(self.base_url)

print("请使用微信扫描二维码登录...")

# 等待登录成功

try:

# 等待直到页面URL改变或出现特定元素

await self.page.wait_for_selector('.feed-card', timeout=60000)

print("登录成功！")

return True

except Exception as e:

print(f"登录超时: {str(e)}")

return False

async def save_cookies(self):

"""保存cookies到文件"""

cookies = await self.context.cookies()

with open(self.cookies_file, 'w') as f:

json.dump(cookies, f)

async def load_cookies(self):

"""从文件加载cookies"""

try:

with open(self.cookies_file, 'r') as f:

cookies = json.load(f)

await self.context.add_cookies(cookies)

return True

except:

return False

async def scroll_and_collect(self, max_videos=10):

"""滚动页面并收集视频"""

video_count = 0

last_height = await self.page.evaluate('document.documentElement.scrollHeight')

while video_count < max_videos:

# 滚动到页面底部

await self.page.evaluate('window.scrollTo(0, document.documentElement.scrollHeight)')

await asyncio.sleep(2) # 等待内容加载

# 获取所有视频元素

video_elements = await self.page.query_selector_all('video')

for video in video_elements:

src = await video.get_attribute('src')

if src:

print(f"找到视频: {src}")

self.save_video_url(src)

video_count += 1

if video_count >= max_videos:

break

# 检查是否到达页面底部

new_height = await self.page.evaluate('document.documentElement.scrollHeight')

if new_height == last_height:

break

last_height = new_height

async def download_video(self, url, filename):

"""下载视频"""

try:

async with aiohttp.ClientSession() as session:

async with session.get(url) as response:

if response.status == 200:

with open(filename, 'wb') as f:

while True:

chunk = await response.content.read(8192)

if not chunk:

break

f.write(chunk)

print(f"成功下载视频: {filename}")

return True

except Exception as e:

print(f"下载视频失败: {str(e)}")

return False

async def process_saved_urls(self):

"""处理保存的视频URL"""

try:

with open("video_urls.txt", "r", encoding="utf-8") as f:

urls = f.readlines()

for i, url in enumerate(urls):

url = url.strip()

if url:

filename = os.path.join(self.download_dir, f"video_{i+1}.mp4")

await self.download_video(url, filename)

except Exception as e:

print(f"处理视频URL失败: {str(e)}")

async def close(self):

"""关闭浏览器"""

if self.browser:

await self.browser.close()

async def main():

scraper = WeixinVideoScraper()

playwright = await scraper.init_browser()

try:

# 尝试加载已保存的cookies

if not await scraper.load_cookies():

# 如果没有cookies，进行登录

if not await scraper.login():

print("登录失败！")

return

# 保存新的cookies

await scraper.save_cookies()

# 导航到视频号页面

await scraper.page.goto(scraper.base_url)

# 输入要收集的视频数量

print("\n请输入要收集的视频数量：")

max_videos = int(input().strip())

# 滚动并收集视频

await scraper.scroll_and_collect(max_videos)

# 下载收集到的视频

await scraper.process_saved_urls()

finally:

await scraper.close()

await playwright.stop()

if __name__ == "__main__":

asyncio.run(main())
